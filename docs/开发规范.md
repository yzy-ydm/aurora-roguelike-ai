## 1. 总体原则

### 1.1 基本原则

| 原则 | 说明 |
|------|------|
| **单一职责** | 每个模块、类、函数只负责一件事 |
| **模块化** | 功能按模块划分，低耦合高内聚 |
| **可读性** | 代码即文档，命名清晰易懂 |
| **渐进开发** | 每次只完成一个明确模块，不一次性生成大型代码 |

### 1.2 禁止事项

- ❌ 一次性生成整个项目代码
- ❌ 未经分析就修改现有代码
- ❌ 随意更改技术路线
- ❌ 提交未测试的代码
- ❌ 在代码中硬编码敏感信息

## 2. Git版本控制规范

### 2.1 分支策略

```
main (主分支)
  │
  ├── develop (开发分支)
  │     │
  │     ├── feature/user-system (功能分支)
  │     ├── feature/game-core (功能分支)
  │     └── feature/ai-generation (功能分支)
  │
  └── hotfix/xxx (热修复分支)
```

| 分支 | 用途 | 命名规则 |
|------|------|----------|
| `main` | 生产环境代码，只接受合并 | - |
| `develop` | 开发主分支 | - |
| `feature/*` | 功能开发 | `feature/功能名称` |
| `hotfix/*` | 紧急修复 | `hotfix/问题描述` |

### 2.2 提交信息规范

格式：`<类型>(<范围>): <描述>`

类型说明：

| 类型 | 说明 | 示例 |
|------|------|------|
| `feat` | 新功能 | `feat(user): 添加用户注册功能` |
| `fix` | 修复bug | `fix(auth): 修复登录验证失败` |
| `docs` | 文档更新 | `docs(api): 更新接口文档` |
| `style` | 代码格式 | `style: 统一代码缩进` |
| `refactor` | 重构 | `refactor(service): 重构AI调用逻辑` |
| `test` | 测试 | `test(auth): 添加登录测试用例` |
| `chore` | 构建/工具 | `chore: 更新依赖版本` |

示例：

```bash
git commit -m "feat(game): 实现随机地图生成算法"
git commit -m "fix(api): 修复token过期未刷新问题"
git commit -m "docs: 添加数据库设计文档"
```

### 2.3 .gitignore 规则

以下文件/目录应被忽略：

- 编辑器配置文件（.vscode, .idea）
- 编译产物（__pycache__, *.pyc）
- 依赖目录（node_modules, venv）
- 环境配置（.env）
- 系�文件（.DS_Store, Thumbs.db）
- Godot缓存（.godot/）

### 3.1 代码风格

遵循 PEP 8 规范：

```python
# ✅ 正确示例
def get_user_by_id(user_id: int) -> User:
    """
    根据用户ID获取用户信息
    
    Args:
        user_id: 用户ID

        User: 用户对象
        
    Raises:
        UserNotFoundError: 用户不存在
```

```python
# app/core/exceptions.py
class AppException(Exception):
    """应用基础异常"""
    def __init__(self, code: int, message: str):
        self.code = code
        self.message = message


class UserNotFoundError(AppException):
    """用户不存在异常"""
    def __init__(self, user_id: int):
        super().__init__(code=404, message=f"用户 {user_id} 不存在")


class AuthenticationError(AppException):
    """认证失败异常"""
    def __init__(self):
        super().__init__(code=401, message="认证失败，请重新登录")
```

```gdscript
# ✅ 正确示例
class_name Player
extends CharacterBody2D

## 玩家角色控制器
##
## 处理玩家的移动、攻击和状态管理

# 信号定义
signal health_changed(new_health: int)
signal died()

# 常量
const MAX_HEALTH: int = 100
const MOVE_SPEED: float = 200.0

# 导出变量
@export var attack_damage: int = 10

# 私有变量
var _current_health: int = MAX_HEALTH
var _is_alive: bool = true


func _ready() -> void:
    """节点初始化完成时调用"""
    _initialize_player()


func _physics_process(delta: float) -> void:
    """物理帧处理，每帧调用"""
    if _is_alive:
        _handle_movement(delta)


## 处理玩家移动
## @param delta: 帧间隔时间
func _handle_movement(delta: float) -> void:
    var velocity = Vector2.ZERO
    
    if Input.is_action_pressed("move_right"):
        velocity.x += 1
    if Input.is_action_pressed("move_left"):
        velocity.x -= 1
    
    velocity = velocity.normalized() * MOVE_SPEED
    self.velocity = velocity
    move_and_slide()
```

### 4.4 HTTP请求封装

```gdscript
# scripts/network/api_client.gd
class_name APIClient
extends Node

## API客户端
##
## 封装所有HTTP请求，统一处理认证和错误

signal request_completed(result: Dictionary)
signal request_failed(error: String)

const BASE_URL: String = "http://localhost:8000/api"

var _http_request: HTTPRequest
var _auth_token: String = ""


func _ready() -> void:
    _http_request = HTTPRequest.new()
    add_child(_http_request)
    _http_request.request_completed.connect(_on_request_completed)


## 发送GET请求
## @param endpoint: API端点路径
## @param params: 查询参数（可选）
func get(endpoint: String, params: Dictionary = {}) -> void:
    var url = BASE_URL + endpoint
    if not params.is_empty():
        url += "?" + _build_query_string(params)
    
    var headers = _build_headers()
    _http_request.request(url, headers, HTTPClient.METHOD_GET)


## 发送POST请求
## @param endpoint: API端点路径
## @param data: 请求体数据
func post(endpoint: String, data: Dictionary = {}) -> void:
    var url = BASE_URL + endpoint
    var headers = _build_headers()
    var json_data = JSON.stringify(data)
    
    _http_request.request(url, headers, HTTPClient.METHOD_POST, json_data)


## 构建请求头
func _build_headers() -> PackedStringArray:
    var headers = PackedStringArray([
        "Content-Type: application/json"
    ])
    
    if not _auth_token.is_empty():
        headers.append("Authorization: Bearer " + _auth_token)
    
    return headers


## 请求完成回调
func _on_request_completed(result: int, response_code: int, headers: PackedStringArray, body: PackedByteArray) -> void:
    if response_code >= 200 and response_code < 300:
        var json = JSON.new()
        json.parse(body.get_string_from_utf8())
        request_completed.emit(json.data)
    else:
        request_failed.emit("Request failed with code: " + str(response_code))
```

## 5. 数据库规范

### 5.1 表命名规范

- 表名使用复数形式：`users`, `games`, `monsters`
- 使用蛇形命名：`game_records`, `ai_generation_logs`
- 关联表使用两个表名：`user_games`, `monster_skills`

### 5.2 字段命名规范

- 主键：`id`
- 外键：`关联表名_id`（如 `user_id`, `game_id`）
- 时间字段：`created_at`, `updated_at`, `deleted_at`
- 布尔字段：`is_活跃状态`（如 `is_active`, `is_deleted`）

### 5.3 SQL示例

```sql
-- 用户表
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    email VARCHAR(100) UNIQUE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_username (username),
    INDEX idx_email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 游戏存档表
CREATE TABLE game_saves (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    save_name VARCHAR(100) NOT NULL,
    game_data JSON NOT NULL,
    floor_level INT DEFAULT 1,
    play_time INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id),
    INDEX idx_user_id (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

## 8. 开发流程

### 8.1 功能开发流程

```
1. 创建功能分支
   git checkout -b feature/功能名称

2. 开发功能
   - 编写代码
   - 添加注释
   - 本地测试

3. 提交代码
   git add .
   git commit -m "feat(模块): 功能描述"

4. 合并到develop
   git checkout develop
   git merge feature/功能名称

5. 删除功能分支
   git branch -d feature/功能名称
```

### 8.2 开发顺序建议

```
Phase 1: 基础框架
├── 服务端框架搭建
├── 数据库连接
└── 客户端基础场景

Phase 2: 用户系统
├── 注册接口
├── 登录接口
└── JWT认证

Phase 3: 游戏核心
├── 地图生成
├── 玩家控制
└── 战斗系统

Phase 4: AI集成
├── AI接口调用
├── 内容生成
└── 结果应用

Phase 5: 完善优化
├── 存档系统
├── UI完善
└── 测试修复
```
